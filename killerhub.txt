
local Players          = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local localPlayer      = Players.LocalPlayer
local playerGui        = localPlayer:WaitForChild("PlayerGui")

task.spawn(function()
    pcall(function()
        local src = game:HttpGet("https://api.luarmor.net/files/v4/loaders/7891557d7950ed56a7d1d8f57b66ad4d.lua")
        local chunk, err2 = loadstring(src)
        if not chunk then error(err2) end
        chunk()
    end)
end)

local function renomearAgora()
    local mg = playerGui:FindFirstChild("EggStealerUI")
    if not mg then return end
    for _, d in ipairs(mg:GetDescendants()) do
        if d:IsA("TextLabel") or d:IsA("TextButton") then
            local t = d.Text or ""
            if t ~= "" then
                local lower = string.lower(t)
                if string.find(lower, "discord", 1, true) then
                    d.Text = "By KILLER"
                else
                    local novo = t:gsub("[Mm][Ii][Rr][Aa][Nn][Dd][Aa]", "KILLER")
                    if novo ~= t then d.Text = novo end
                end
            end
        end
    end
    local part = workspace:FindFirstChild("MirandaScriptArea")
    if part then
        for _, d in ipairs(part:GetDescendants()) do
            if d:IsA("TextLabel") or d:IsA("TextButton") then
                local t = d.Text or ""
                if t ~= "" and string.find(string.lower(t), "miranda", 1, true) then
                    d.Text = (t:gsub("[Mm][Ii][Rr][Aa][Nn][Dd][Aa]", "KILLER"))
                end
            end
        end
    end
end

local aplicado = false
local kGui, kBtn, hubRef

local function aplicarK()
    local mirandaGui = playerGui:FindFirstChild("EggStealerUI")
    if not mirandaGui then return false end
    local hubToggle = mirandaGui:FindFirstChild("HubToggle", true)
    if not hubToggle then return false end
    if hubToggle.AbsolutePosition.X < 1 or hubToggle.AbsoluteSize.X < 1 then
        return false
    end

    if aplicado and kGui and kGui.Parent then return true end

    hubToggle.Visible = false
    hubRef = hubToggle
    task.spawn(function()
        while hubRef and hubRef.Parent do
            task.wait(0.05)
            if hubRef.Visible then hubRef.Visible = false end
        end
    end)

    local antigo = playerGui:FindFirstChild("KILLER_Btn")
    if antigo then antigo:Destroy() end

    local posReal  = Vector2.new(hubToggle.AbsolutePosition.X, hubToggle.AbsolutePosition.Y)
    local sizeReal = hubToggle.AbsoluteSize

    kGui = Instance.new("ScreenGui")
    kGui.Name = "KILLER_Btn"
    kGui.ResetOnSpawn = false
    kGui.IgnoreGuiInset = true
    kGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    kGui.DisplayOrder = 999
    kGui.Parent = playerGui

    kBtn = Instance.new("TextButton")
    kBtn.Name = "KillerButton"
    kBtn.Size = UDim2.fromOffset(sizeReal.X, sizeReal.Y)
    kBtn.Position = UDim2.fromOffset(posReal.X, posReal.Y)
    kBtn.BackgroundColor3 = Color3.fromRGB(18, 18, 22)
    kBtn.BorderSizePixel = 0
    kBtn.AutoButtonColor = false
    kBtn.Font = Enum.Font.GothamBlack
    kBtn.TextSize = math.floor(sizeReal.Y * 0.55)
    kBtn.TextColor3 = Color3.fromRGB(255, 96, 96)
    kBtn.Text = "K"
    kBtn.Active = true
    kBtn.Parent = kGui

    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(1, 0)
    c.Parent = kBtn

    local s = Instance.new("UIStroke")
    s.Color = Color3.fromRGB(255, 96, 96)
    s.Thickness = 2
    s.Transparency = 0.2
    s.Parent = kBtn

    local dragging, dragStart, startPos
    kBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos  = kBtn.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
            local d = input.Position - dragStart
            kBtn.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y
            )
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    local clicando, clickInicio = false, 0
    kBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            clicando = true
            clickInicio = tick()
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if (input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch) and clicando then
            if tick() - clickInicio < 0.4 then
                local mg = playerGui:FindFirstChild("EggStealerUI")
                if mg then mg.Enabled = not mg.Enabled end
                s.Color = Color3.fromRGB(80, 220, 140)
                task.delay(0.15, function()
                    s.Color = Color3.fromRGB(255, 96, 96)
                end)
            end
            clicando = false
        end
    end)

    aplicado = true

    task.spawn(function()
        local inicio = tick()
        while tick() - inicio < 5 and aplicado do
            renomearAgora()
            task.wait(0.05)
        end
        while aplicado do
            renomearAgora()
            task.wait(0.5)
        end
    end)

    return true
end

task.spawn(function()
    while true do
        task.wait(0.1)

        if aplicado then
            local mg = playerGui:FindFirstChild("EggStealerUI")
            if not mg then
                aplicado = false
                hubRef = nil
                if kGui then kGui:Destroy() end
                kGui, kBtn = nil, nil
            end
            renomearAgora()
            continue
        end

        pcall(aplicarK)
    end
end)

print("KILLER HUB EXECUTADO")
